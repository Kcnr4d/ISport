abstract class Uzytkownik {
    # id : int
    # imie : String
    # nazwisko : String
    # email : String
    # hasloHash : String
    # telefon : String
    + zaloguj(email, haslo) : boolean
    + wyloguj() : void
    + zmienHaslo(stare, nowe) : boolean
}

class Klient extends Uzytkownik {
    + zarejestruj() : void
    + przegladajSprzet() : List<Sprzet>
    + utworzRezerwacje(sprzet, okres) : Rezerwacja
    + anulujRezerwacje(rezerwacja) : void
    + pobierzRabaty() : List<Rabat>
}

class PracownikBiura extends Uzytkownik {
    - rola : String
    - uprawnienia : List<String>
    + dodajSprzet(dane) : Sprzet
    + edytujSprzet(sprzet, dane) : void
    + usunSprzet(sprzet) : boolean
    + oznaczZwrot(wypozyczenie) : void
    + przyznajRabat(klient, rabat) : void
    + wyslijWiadomosc(klient, tresc) : void
    + generujCennik(zestaw, okres) : Oferta
}

class Sprzet {
    - id : int
    - nazwa : String
    - opis : String
    - cenaZaDobe : Decimal
    - stanTechniczny : String
    - status : StatusSprzetu
    + zmienStatus(status) : void
    + czyDostepny(okres) : boolean
    + aktualizujDane(dane) : void
}

class Kategoria {
    - id : int
    - nazwa : String
    - opis : String
}

class Rezerwacja {
    - id : int
    - dataUtworzenia : DateTime
    - dataRozpoczecia : Date
    - dataZakonczenia : Date
    - status : StatusRezerwacji
    - kosztCalkowity : Decimal
    + obliczKoszt() : Decimal
    + zatwierdz() : void
    + anuluj() : void
    + aktualizujStatus(status) : void
}

class Wypozyczenie {
    - id : int
    - dataWydania : DateTime
    - dataZwrotuPlanowana : Date
    - dataZwrotuFaktyczna : DateTime
    - status : StatusWypozyczenia
    + zakoncz() : void
    + czyPrzeterminowane() : boolean
}

class Platnosc {
    - id : int
    - kwota : Decimal
    - kwotaPoRabacie : Decimal
    - dataPlatnosci : DateTime
    - status : String
    + przelicz() : void
}

class Rabat {
    - id : int
    - typ : TypRabatu
    - wartosc : Decimal
    - dataPrzyznania : DateTime
    - przyznanyPrzez : String
    - aktywny : boolean
    + zastosuj(kwota) : Decimal
}

class Oferta {
    - id : int
    - dataWygenerowania : DateTime
    - kosztLaczny : Decimal
}

class Wiadomosc {
    - id : int
    - tresc : String
    - kanal : KanalKomunikacji
    - dataWyslania : DateTime
    - dostarczona : boolean
}

class Powiadomienie {
    - id : int
    - tresc : String
    - typ : String
    - dataUtworzenia : DateTime
    + wyslij() : void
}

enum StatusSprzetu {
    DOSTEPNY
    WYPOZYCZONY
    ZAREZERWOWANY
    NIEDOSTEPNY
}

enum StatusRezerwacji {
    OCZEKUJACA
    POTWIERDZONA
    AKTYWNA
    ZAKONCZONA
    ANULOWANA
}

enum StatusWypozyczenia {
    AKTYWNE
    ZAKONCZONE
}

enum TypRabatu {
    PROCENTOWY
    KWOTOWY
}

enum KanalKomunikacji {
    EMAIL
    PUSH
}
