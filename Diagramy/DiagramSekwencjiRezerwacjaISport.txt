@startuml ISport_Sekwencja_Rezerwacja
title Diagram sekwencji - Rezerwacja sprzętu (HS-16, HS-17, HS-18, HS-15)

skinparam backgroundColor #FEFEFE
skinparam sequence {
    ArrowColor #185FA5
    LifeLineBorderColor #5F5E5A
    LifeLineBackgroundColor #F1EFE8
    ParticipantBorderColor #185FA5
    ParticipantBackgroundColor #E6F1FB
    ParticipantFontColor #042C53
    ActorBorderColor #534AB7
    ActorBackgroundColor #EEEDFE
}
skinparam noteBackgroundColor #FAEEDA
skinparam noteBorderColor #BA7517

actor "Klient" as K
participant "Aplikacja\nmobilna" as App
participant "System\nISport" as Sys
participant "Moduł\nsprzętu" as MS
participant "Moduł\nrezerwacji" as MR
participant "Moduł\npłatności" as MP
participant "Baza\ndanych" as DB

== Przeglądanie sprzętu (HS-16) ==

K -> App : przechodzi do sekcji przeglądania sprzętu
activate App
App -> Sys : pobierzDostepnySprzet(filtry)
activate Sys
Sys -> MS : znajdzSprzet(status=dostepny, filtry)
activate MS
MS -> DB : SELECT sprzęt
activate DB
DB --> MS : lista sprzętu
deactivate DB
MS --> Sys : lista dostępnego sprzętu
deactivate MS
Sys --> App : lista sprzętu
deactivate Sys
App --> K : wyświetla sprzęt z nazwą, ceną

K -> App : wybiera sprzęt

== Sprawdzanie dostępności i wybór dat (HS-17, HS-18) ==

App -> Sys : pobierzDostepnoscTerminow(sprzetId)
activate Sys
Sys -> MR : sprawdzZajeteTerminy(sprzetId)
activate MR
MR -> DB : SELECT rezerwacje dla sprzętu
activate DB
DB --> MR : zajęte terminy
deactivate DB
MR --> Sys : harmonogram dostępności
deactivate MR
Sys --> App : kalendarz z zablokowanymi dniami
deactivate Sys
App --> K : wyświetla kalendarz

K -> App : wybiera datę rozpoczęcia i zakończenia

== Kalkulacja kosztu (HS-15) ==

App -> Sys : obliczKoszt(sprzetId, dataOd, dataDo)
activate Sys
Sys -> MP : kalkulujKoszt(sprzet, liczbaDni)
activate MP
MP -> DB : pobierzCeneSprzetu(sprzetId)
activate DB
DB --> MP : cena za dobę
deactivate DB
MP -> DB : pobierzAktywnyRabat(klientId)
activate DB
DB --> MP : rabat (jeśli istnieje)
deactivate DB
MP -> MP : kosztBazowy = cena × liczbaDni
MP -> MP : kosztKoncowy = zastosujRabat(kosztBazowy)
MP --> Sys : podsumowanie kosztu
deactivate MP
Sys --> App : koszt przed i po rabacie
deactivate Sys
App --> K : wyświetla podsumowanie\n(cena/doba, koszt całkowity, rabat)

== Zatwierdzenie rezerwacji ==

alt klient akceptuje koszt
    K -> App : potwierdza rezerwację
    App -> Sys : utworzRezerwacje(klientId, sprzetId, okres, koszt)
    activate Sys

    Sys -> MR : sprawdzPonownieDostepnosc(sprzetId, okres)
    activate MR
    MR -> DB : weryfikacja kolizji terminów
    activate DB
    DB --> MR : brak kolizji
    deactivate DB

    MR -> DB : INSERT rezerwacja(status=oczekujaca)
    activate DB
    DB --> MR : rezerwacjaId
    deactivate DB

    MR -> MS : zmienStatusSprzetu(sprzetId, zarezerwowany)
    activate MS
    MS -> DB : UPDATE status sprzętu
    activate DB
    DB --> MS : OK
    deactivate DB
    MS --> MR : status zaktualizowany
    deactivate MS

    MR --> Sys : rezerwacja utworzona
    deactivate MR
    Sys --> App : potwierdzenie rezerwacji
    deactivate Sys
    App --> K : "Rezerwacja potwierdzona"
    deactivate App
else klient cofa się do edycji
    K -> App : wybiera "Wstecz"
    App --> K : powrót do wyboru dat
    deactivate App
end

@enduml
